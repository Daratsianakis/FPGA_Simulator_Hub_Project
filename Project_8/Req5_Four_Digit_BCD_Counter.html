EXPLORE / STUDY / VERIFYRTL + DOCUMENTATION
  • Four BCD counters increment as a decimal chain. D0 advances every 1 kHz edge; D1 advances when D0 is 9; D2 advances when D0 and D1 are 9; D3 advances when the lower three are all 9.
  • The four-digit count increments 1,000 times per second and wraps after 10,000 increments, corresponding to ten seconds.
  • A 2-bit scan counter selects a digit through a 4:1 bus mux. A 2:4 decoder selects the active-low digit enable, while the BCD decoder drives the cathodes.
  • The display is rapidly multiplexed; the visual simulator presents the persistent four-digit number for readability.
1,000 BCD increments/s × 10 s = 10,000 increments; scan = 1 kHz / 4 = 250 Hz per digit
Browser time is accelerated for learning. The display shows corresponding sampled logic states, not every 10 ns master-clock transition.

// Requirement 5: 0000..9999 in ten seconds at 1000 increments/sec.
module req5_four_digit_bcd (
    input wire clk, output wire [3:0] seg_an, output wire [7:0] seg_cat
);
    wire clk_1khz;
    wire [3:0] d0, d1, d2, d3;
    p8_clock_divider #(.HALF_PERIOD(50000)) divider_1khz (
        .clk(clk), .clk_out(clk_1khz)
    );
    p8_bcd_digit digit0 (.clk(clk_1khz), .enable(1'b1),
        .load(1'b0), .load_value(4'b0000), .q(d0));
    p8_bcd_digit digit1 (.clk(clk_1khz), .enable(d0 == 9),
        .load(1'b0), .load_value(4'b0000), .q(d1));
    p8_bcd_digit digit2 (.clk(clk_1khz), .enable((d0 == 9) && (d1 == 9)),
        .load(1'b0), .load_value(4'b0000), .q(d2));
    p8_bcd_digit digit3 (.clk(clk_1khz), .enable((d0 == 9) && (d1 == 9) && (d2 == 9)),
        .load(1'b0), .load_value(4'b0000), .q(d3));
    p8_sevenseg_scan scan (.clk_1khz(clk_1khz),
        .d0(d0), .d1(d1), .d2(d2), .d3(d3), .seg_an(seg_an), .seg_cat(seg_cat));
endmodule


// Reusable 50-percent-duty clock divider. Output frequency = input / (2 * HALF_PERIOD).
module p8_clock_divider #(parameter integer HALF_PERIOD = 50000) (
    input wire clk, output reg clk_out = 0
);
    localparam integer WIDTH = (HALF_PERIOD > 1) ? $clog2(HALF_PERIOD) : 1;
    reg [WIDTH-1:0] count = 0;
    always @(posedge clk) begin
        if (count == HALF_PERIOD - 1) begin
            count <= 0;
            clk_out <= ~clk_out;
        end else count <= count + 1;
    end
endmodule


// Synchronous BCD digit with enable and optional parallel load.
module p8_bcd_digit (
    input wire clk, input wire enable, input wire load,
    input wire [3:0] load_value, output reg [3:0] q = 0
);
    always @(posedge clk) begin
        if (load) q <= (load_value > 9) ? 9 : load_value;
        else if (enable) q <= (q == 9) ? 0 : q + 1;
    end
endmodule


// 1kHz scanning clock selects each of four digits in turn (250Hz per digit).
module p8_sevenseg_scan (
    input wire clk_1khz,
    input wire [3:0] d0, d1, d2, d3,
    output wire [3:0] seg_an,
    output wire [7:0] seg_cat
);
    reg [1:0] scan = 0;
    wire [3:0] current_digit;
    wire [6:0] segments;
    always @(posedge clk_1khz) scan <= scan + 1;
    p8_mux4x4 bus_mux (.d0(d0), .d1(d1), .d2(d2), .d3(d3), .sel(scan), .y(current_digit));
    p8_decoder2to4 anode_decode (.sel(scan), .seg_an(seg_an));
    p8_bcd7_decoder cathode_decode (.bcd(current_digit), .seg_cat(segments));
    assign seg_cat = {1'b1, segments}; // DP off
endmodule


// Explicit 4:1, four-bit bus multiplexer.
module p8_mux4x4 (
    input wire [3:0] d0, d1, d2, d3, input wire [1:0] sel,
    output reg [3:0] y
);
    always @* begin
        case (sel)
            0: y = d0;
            1: y = d1;
            2: y = d2;
            3: y = d3;
        endcase
    end
endmodule


// 2:4 decoder: digit enables on Blackboard are active low.
module p8_decoder2to4 (input wire [1:0] sel, output wire [3:0] seg_an);
    assign seg_an = ~(4'b0001 << sel);
endmodule


// Blackboard cathodes seg_cat[0]=A .. seg_cat[6]=G. Active low.
module p8_bcd7_decoder (input wire [3:0] bcd, output reg [6:0] seg_cat);
    always @* begin
        case (bcd)
            0: seg_cat = 7'b1000000;
            1: seg_cat = 7'b1111001;
            2: seg_cat = 7'b0100100;
            3: seg_cat = 7'b0110000;
            4: seg_cat = 7'b0011001;
            5: seg_cat = 7'b0010010;
            6: seg_cat = 7'b0000010;
            7: seg_cat = 7'b1111000;
            8: seg_cat = 7'b0000000;
            9: seg_cat = 7'b0010000;
            default: seg_cat = 7'b1111111;
        endcase
    end
endmodule

How to operate this requirement

  1. Click Step +0.01s and inspect the four BCD digits in the architecture diagram.
  2. Observe carries at 0009 → 0010, 0099 → 0100, and 0999 → 1000.
  3. The 2-bit scan signal selects which BCD value is decoded at that instant.
  4. At 10 seconds the counter cycles back to 0000.

Hardware notes

The virtual controls are a learning model; actual FPGA operation uses the 100 MHz Blackboard clock and the top-level Verilog module. Run, Step and Reset are browser simulation controls, not additional physical board inputs.

Browser time is accelerated for learning. The display shows corresponding sampled logic states, not every 10 ns master-clock transition.
