DESIGN EXPLANATIONREQ 3

  • The first toggle divider changes state every 50,000 input clock cycles, producing a full 1 kHz waveform.
  • The second changes state every 500 edges of the 1 kHz waveform, producing a 1 Hz waveform.
  • The LED is high for 0.5 seconds and low for 0.5 seconds.
  • A real LED is driven by the full-speed hardware. The webpage uses scaled elapsed-time simulation.

// Requirement 3: 100MHz -> 1kHz -> 1Hz; LED blinks at one full cycle per second.
module req3_cascaded_dividers (input wire clk, output wire [7:0] led);
    wire clk_1khz;
    wire clk_1hz;
    p8_clock_divider #(.HALF_PERIOD(50000)) divide_to_1khz (
        .clk(clk), .clk_out(clk_1khz)
    );
    p8_clock_divider #(.HALF_PERIOD(500)) divide_to_1hz (
        .clk(clk_1khz), .clk_out(clk_1hz)
    );
    assign led = {7'b0000000, clk_1hz};
endmodule


// Reusable 50-percent-duty clock divider. Output frequency = input / (2 * HALF_PERIOD).
module p8_clock_divider #(parameter integer HALF_PERIOD = 50000) (
    input wire clk, output reg clk_out = 0
);
    localparam integer WIDTH = (HALF_PERIOD > 1) ? $clog2(HALF_PERIOD) : 1;
    reg [WIDTH-1:0] count = 0;
    always @(posedge clk) begin
        if (count == HALF_PERIOD - 1) begin
            count <= 0;
            clk_out <= ~clk_out;
        end else count <= count + 1;
    end
endmodule
